# Trailing space at end of line
\s+$

# Double hyphen instead of em dash
--(?!-)

# Straight quotes (check for curly conversion)
["']

# Mixed languages (basic Latin + Cyrillic)
(?=.*[A-Za-z])(?=.*[А-Яа-я])

# Placeholder not preserved (e.g. {0})
\{[0-9]+\}(?!.*\{[0-9]+\})

# Strip tags/markers (e.g. `<span>`, `<br />`, etc.)
`<[^>]+>`
